package __RASD_PACKAGE__;

import android.app.Activity;
import android.content.Intent;
import android.net.Uri;
import android.widget.Toast;
import androidx.activity.result.ActivityResult;
import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.ActivityCallback;
import com.getcapacitor.annotation.CapacitorPlugin;
import java.io.File;
import java.io.FileInputStream;
import java.io.InputStream;
import java.io.OutputStream;
import java.io.IOException;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;

@CapacitorPlugin(name = "RasdDocumentSave")
public class RasdDocumentSavePlugin extends Plugin {
    private static final long MAX_FILE_BYTES = 25L * 1024 * 1024;
    // Files in app cache ONLY, generated by Rasd's encrypted backup exporter.
    private static final String FILENAME = "^rasd-android-[A-Za-z0-9-]{10,100}\\.json$";

    @PluginMethod
    public void saveEncryptedDocument(PluginCall call) {
        final String fileName = call.getString("fileName");
        final String expectedSha256 = call.getString("sha256");
        if (fileName == null || !fileName.matches(FILENAME)
                || expectedSha256 == null || !expectedSha256.matches("^[a-f0-9]{64}$")) {
            call.reject("Invalid encrypted backup file name or digest");
            return;
        }
        try {
            File cacheFile = new File(getContext().getCacheDir(), fileName);
            if (!cacheFile.getCanonicalFile().getParentFile().equals(getContext().getCacheDir().getCanonicalFile())
                || !cacheFile.isFile() || cacheFile.length() < 1 || cacheFile.length() > MAX_FILE_BYTES) {
                call.reject("Encrypted backup cache file unavailable");
                return;
            }
            if (!digestFile(cacheFile).equals(expectedSha256)) {
                call.reject("Encrypted backup integrity mismatch before document picker");
                return;
            }
        } catch (Exception ex) {
            call.reject("Cannot inspect encrypted backup file");
            return;
        }
        Intent intent = new Intent(Intent.ACTION_CREATE_DOCUMENT);
        intent.addCategory(Intent.CATEGORY_OPENABLE);
        intent.setType("application/json");
        intent.putExtra(Intent.EXTRA_TITLE, fileName);
        startActivityForResult(call, intent, "onDocumentChosen");
    }

    @ActivityCallback
    private void onDocumentChosen(PluginCall call, ActivityResult result) {
        if (call == null) return;
        if (result.getResultCode() != Activity.RESULT_OK || result.getData() == null
                || result.getData().getData() == null) {
            call.reject("تم إلغاء حفظ النسخة أو لم يختر المستخدم وجهة");
            return;
        }
        final Uri destination = result.getData().getData();
        final String filename = call.getString("fileName");
        final String expectedDigest = call.getString("sha256");
        // OS-selected URI is the ONLY write destination. No storage permission,
        // direct Google account access, or potentially secret content is sent.
        new Thread(() -> {
            try {
                File source = new File(getContext().getCacheDir(), filename);
                if (!source.getCanonicalFile().getParentFile().equals(getContext().getCacheDir().getCanonicalFile())
                    || !source.isFile() || source.length() < 1 || source.length() > MAX_FILE_BYTES
                    || !digestFile(source).equals(expectedDigest)) {
                    throw new IOException("Source backup changed before save");
                }
                try (InputStream input = new FileInputStream(source);
                     OutputStream output = getContext().getContentResolver().openOutputStream(destination, "w")) {
                    if (output == null) throw new IOException("Selected document provider returned null");
                    byte[] buffer = new byte[65536];
                    int read;
                    long total = 0;
                    while ((read = input.read(buffer)) != -1) {
                        total += read;
                        if (total > MAX_FILE_BYTES) throw new IOException("File exceeds 25 MB");
                        output.write(buffer, 0, read);
                    }
                    output.flush();
                }
                getActivity().runOnUiThread(() -> {
                    call.resolve(new JSObject().put("saved", true));
                    Toast.makeText(getContext(), "حُفظت النسخة المشفرة في المكان المختار", Toast.LENGTH_LONG).show();
                });
            } catch (Exception ex) {
                getActivity().runOnUiThread(() -> call.reject("تعذر كتابة النسخة في الوجهة المختارة"));
            }
        }, "rasd-encrypted-document-save").start();
    }

    private static String digestFile(File file) throws IOException, NoSuchAlgorithmException {
        MessageDigest digest = MessageDigest.getInstance("SHA-256");
        try (InputStream input = new FileInputStream(file)) {
            byte[] buffer = new byte[65536];
            int read;
            while ((read = input.read(buffer)) != -1) digest.update(buffer, 0, read);
        }
        StringBuilder result = new StringBuilder(64);
        for (byte b : digest.digest()) result.append(String.format(java.util.Locale.ROOT, "%02x", b & 0xff));
        return result.toString();
    }
}
